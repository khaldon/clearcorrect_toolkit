/**
 * Retrieves the value for 'Engagers removal for revisions' from the page.
 * @returns {string|null} The value found, or null if not found.
 */
function getEngagersRemovalValue() {
  const rows = Array.from(document.querySelectorAll('tr'));
  const targetRow = rows.find(tr => {
    const cells = tr.querySelectorAll('td');
    // Ensure the row has at least 2 cells and the first one matches our label
    return cells.length >= 2 && cells[0].innerText.trim() === 'Engagers removal for revisions';
  });

  return targetRow ? targetRow.cells[1].innerText.trim() : null;
}

const value = getEngagersRemovalValue();
console.log("Engagers removal for revisions:", value);

(async function findOnlineForm() {
    const ONLINE_FORM_TAB_ID = 'ctl00_MainPH_frmCaseSubmission_tcRecord_tpForm_tab';
    const FILES_TAB_ID = 'ctl00_MainPH_frmCaseSubmission_tcRecord_tpFiles_tab';
    const VERSION_DROPDOWN_ID = 'ctl00_MainPH_frmCaseSubmission_tcRecord_tpFiles_ddlCaseSubmissionID';

    const onlineFormTab = document.getElementById(ONLINE_FORM_TAB_ID);
    
    // 1. Check if "Online Form" tab is present and visible
    // In ASP.NET AJAX Tabs, the tab might exist but be hidden or empty
    const isOnlineFormAvailable = onlineFormTab && onlineFormTab.offsetHeight > 0;

    if (isOnlineFormAvailable) {
        console.log("✅ Online Form found! Switching to tab...");
        onlineFormTab.click();
        return;
    }

    console.log("❌ Online Form not found in this version. Navigating to Files to change version...");

    // 2. Go to Files tab to access the dropdown
    const filesTab = document.getElementById(FILES_TAB_ID);
    if (!filesTab) {
        console.error("Could not find Files tab.");
        return;
    }
    
    // Click Files tab if not already there
    filesTab.click();

    // Small delay to ensure the Files content is visible (or wait for the dropdown)
    setTimeout(() => {
        const dropdown = document.getElementById(VERSION_DROPDOWN_ID);
        if (!dropdown) {
            console.error("Version dropdown not found in Files tab.");
            return;
        }

        const currentIndex = dropdown.selectedIndex;
        
        // 3. Go to the previous version in the dropdown
        // Note: In your dropdown, the "Revision" headers and the actual dates are mixed.
        // We move to the next item in the list (older version).
        if (currentIndex < dropdown.options.length - 1) {
            console.log(`Switching from version ${dropdown.options[currentIndex].text} to older version...`);
            dropdown.selectedIndex = currentIndex + 1;
            
            // Trigger the change event to force the page to reload/update
            dropdown.dispatchEvent(new Event('change'));
            
            // If it's a standard ASP.NET postback, the __doPostBack function might be needed:
            if (typeof __doPostBack !== 'undefined') {
                __doPostBack(VERSION_DROPDOWN_ID, '');
            }
        } else {
            console.log("Reached the end of the version history. Online Form not found.");
        }
    }, 500);
})();

/**
 * Finds the last table row containing a submission link and clicks it.
 */
function clickLastSubmissionLink() {
  // 1. Get all table rows
  const allRows = Array.from(document.querySelectorAll('tr'));

  // 2. Filter for rows that actually contain the submission button/image
  const rowsWithLink = allRows.filter(row => 
    row.querySelector('.btnlgray.bdlgray.hand img')
  );

  if (rowsWithLink.length > 0) {
    // 3. Select the last row in the filtered list
    const lastRow = rowsWithLink[rowsWithLink.length - 1];
    
    // 4. Find the image within that last row and click it
    const targetImage = lastRow.querySelector('.btnlgray.bdlgray.hand img');
    
    if (targetImage) {
      targetImage.click();
      console.log("Successfully clicked the submission link in the last row.");
    }
  } else {
    console.error("No rows found with the submission link.");
  }
}

clickLastSubmissionLink();
